Configurar backup automático BACKUP 
import shutil
from pathlib import Path

BACKUP_EVERY = 5
WEIGHTS_DIR  = Path('/content/runs/finetune_v2/weights')
BACKUP_DIR   = Path(DRIVE_BACKUP_DIR)

def backup_a_drive(trainer):
    epoca = trainer.epoch + 1
    if epoca % BACKUP_EVERY != 0:
        return
    try:
        for nombre in ['last.pt', 'best.pt']:
            origen = WEIGHTS_DIR / nombre
            if origen.exists():
                shutil.copy2(origen, BACKUP_DIR / f'epoch{epoca:03d}_{nombre}')
                shutil.copy2(origen, BACKUP_DIR / nombre)
        print(f'  [BACKUP] Época {epoca} → Drive ✓')
    except Exception as e:
        print(f'  [BACKUP] Error: {e}')

print(f'Backup configurado: cada {BACKUP_EVERY} épocas → {BACKUP_DIR}')

🚀 Entrenamiento desde cero — Dataset limpio

El modelo aprende sin sesgos acumulados. Requiere dataset limpio de Roboflow (sin labels de segmentación mezcladas) y ~8-10 horas en T4.

📋 Antes de empezar — Limpiar dataset en Roboflow

  1. En Roboflow → tu proyecto → Generate New Version
  2. En Preprocessing: activar Stratified Split (balancea excavadora en validación)
  3. En Export: seleccionar YOLOv8 → Object Detection (NO Instance Segmentation)
  4. Descargar el zip y subirlo a Drive como dataset_limpio.zip
  5. Subir también el archivo yolo11n.pt base (se descarga automáticamente en Colab)
⚠️ El 9% del dataset actual tiene labels de segmentación mezcladas con detección. Exportar como "Object Detection" las elimina automáticamente.
ℹ️ Las celdas 1, 2, 4, 5, 6 son iguales al fine-tuning. La celda 3 cambia (no necesita best.pt). La celda 7 no aplica (no hay checkpoint previo).

from ultralytics import YOLO
import torch

if 'BATCH' not in dir():
    vram = torch.cuda.get_device_properties(0).total_memory / 1024**3 if torch.cuda.is_available() else 0
    BATCH = 16 if vram >= 14 else (8 if vram >= 8 else 4)
    print(f'BATCH: {BATCH}')

# Cargar YOLO11n base preentrenado en COCO
# Se descarga automáticamente ~6MB — NO es tu best.pt de minería
model = YOLO('yolo11n.pt')
model.add_callback('on_train_epoch_end', backup_a_drive)

print('=' * 60)
print('  Entrenamiento DESDE CERO — YOLO11n + Dataset Limpio')
print('=' * 60)
print(f'  Modelo base : yolo11n.pt (COCO pretrained — Ultralytics)')
print(f'  Dataset     : {DATA_YAML}')
print(f'  Épocas      : 100')
print(f'  lr0         : 0.01 (tasa normal)')
print(f'  Batch       : {BATCH}')
print(f'  Backup      : cada 5 épocas → {DRIVE_BACKUP_DIR}')
print('  Resultado   : runs/nano_scratch_v1/weights/best.pt')
print('=' * 60)

results = model.train(
    data          = DATA_YAML,
    epochs        = 100,        # desde cero necesita más épocas que fine-tuning
    imgsz         = 640,
    batch         = BATCH,
    project       = '/content/runs',
    name          = 'nano_scratch_v1',
    exist_ok      = True,
    resume        = False,      # NUNCA hacer resume desde cero

    # Tasa de aprendizaje NORMAL — no conservadora como fine-tuning
    optimizer     = 'SGD',
    lr0           = 0.01,       # 333x más alta que fine-tuning (0.00003)
    lrf           = 0.01,
    momentum      = 0.937,
    weight_decay  = 0.0005,
    warmup_epochs = 5,          # más warmup para partir limpio

    # Sin congelamiento — todo entrenable desde cero
    freeze        = 0,
    patience      = 30,         # más paciencia — aprende más lento al inicio

    # Augmentación agresiva — aprende variedad desde la época 1
    degrees       = 20.0,
    perspective   = 0.002,
    shear         = 5.0,
    fliplr        = 0.5,
    flipud        = 0.15,
    scale         = 0.8,
    mosaic        = 1.0,
    copy_paste    = 0.5,
    erasing       = 0.4,
    mixup         = 0.15,       # un poco más que fine-tuning
    hsv_h         = 0.015,
    hsv_s         = 0.6,        # más saturación — variedad de condiciones
    hsv_v         = 0.4,
    amp           = True,
    save          = True,
    save_period   = 5,
    workers       = 2,
    verbose       = True,
)
